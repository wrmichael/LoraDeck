cmake_minimum_required(VERSION 3.16)
file(READ "${CMAKE_CURRENT_LIST_DIR}/version.txt" PROJECT_VER)
string(STRIP "${PROJECT_VER}" PROJECT_VER)
# Run the configure step again when version.txt changes; otherwise the old number stays in the firmware.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/version.txt")
# Board: -DLORADECK_BOARD=heltec_v3 (default) or cardputer_adv. Each board has its own project name, so the
# web update refuses firmware for the other board, and its own sdkconfig additions.
if(NOT DEFINED LORADECK_BOARD)
    set(LORADECK_BOARD heltec_v3)
endif()
if(LORADECK_BOARD STREQUAL "cardputer_adv")
    set(LD_PROJECT loradeck-cardputer)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.cardputer")
    add_compile_definitions(LORADECK_CARDPUTER=1)
elseif(LORADECK_BOARD STREQUAL "heltec_v3")
    set(LD_PROJECT loradeck)
else()
    message(FATAL_ERROR "LORADECK_BOARD must be heltec_v3 or cardputer_adv")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${LD_PROJECT})
